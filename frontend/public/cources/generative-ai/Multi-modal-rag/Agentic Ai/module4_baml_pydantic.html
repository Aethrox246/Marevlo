PYDANTIC BASICS

Pydantic is Python's most popular data validation library. It turns Python type annotations into runtime validators — and auto-generates JSON Schema for free.

Pydantic lets you define what your data should look like using Python classes and type hints. When you create an instance, Pydantic automatically validates every field — wrong type, wrong value, missing field — all caught at construction time.

Python — Your First Pydantic Model
# pip install pydantic
from pydantic import BaseModel, Field
from typing import Literal, Optional

# Define your data shape as a class
class DeviceReport(BaseModel):
    device_id:  str
    risk_level: Literal["LOW", "MED", "HIGH"]
    score:      float = Field(ge=0.0, le=1.0)
    root_cause: str
    actions:    list[str] = Field(min_length=1)

# ✓ Valid — constructs successfully
report = DeviceReport(
    device_id  = "JNP-001",
    risk_level = "HIGH",
    score      = 0.91,
    root_cause = "rpd memory leak recurrence",
    actions    = ["Restart rpd", "Check BGP peers"]
)
print(report.risk_level)   # "HIGH"
print(report.score)        # 0.91
print(report.model_dump()) # Python dict
print(report.model_dump_json()) # JSON string

# ✗ Invalid — raises ValidationError immediately
try:
    bad = DeviceReport(
        device_id  = "JNP-001",
        risk_level = "CRITICAL",  # Not in Literal!
        score      = 1.5,           # Exceeds le=1.0!
        root_cause = "unknown",
        actions    = []               # Violates min_length=1!
    )
except Exception as e:
    print(f"Caught {len(e.errors())} errors")

Nested Models & Type Coercion

Pydantic handles nested models automatically. It also coerces compatible types — "94.2" (string) becomes 94.2 (float) — which is critical for LLM output parsing where types are often stringified.

Python — Nested Pydantic Models
from pydantic import BaseModel, Field
from typing import Literal, Optional
from datetime import datetime

class Metric(BaseModel):
    name:        str
    value:       float
    unit:        str
    is_anomalous: bool

class RemediationStep(BaseModel):
    order:              int = Field(ge=1)
    action:             str = Field(min_length=10)
    estimated_minutes:  int = Field(ge=1, le=480)
    requires_downtime:  bool = False

class ComprehensiveReport(BaseModel):
    device_id:    str
    timestamp:    datetime
    risk_level:   Literal["LOW", "MED", "HIGH"]
    score:        float = Field(ge=0, le=1)
    metrics:      list[Metric]
    steps:        list[RemediationStep]
    notes:        Optional[str] = None

# Pydantic coerces types automatically
data_from_llm = {
    "device_id":  "JNP-001",
    "timestamp":  "2024-03-15T06:00:00Z",  # str → datetime ✓
    "risk_level": "HIGH",
    "score":      "0.91",                   # str → float ✓
    "metrics": [
        {"name": "cpu_util", "value": "94.2", "unit": "%", "is_anomalous": "true"}
    ],
    "steps": [
        {"order": "1", "action": "Restart the rpd routing process", "estimated_minutes": "5"}
    ]
}

report = ComprehensiveReport.model_validate(data_from_llm)
print(f"Score type: {type(report.score)}")         # float
print(f"Timestamp: {type(report.timestamp)}")       # datetime
print(f"CPU anomalous: {report.metrics[0].is_anomalous}")  # True (bool)
⚡
Type coercion is your friend: LLMs frequently return numbers as strings ("0.91" instead of 0.91). Pydantic's model_validate() coerces compatible types automatically. Use model_validate() (not the constructor directly) for LLM data.

Python — High-Performance Pydantic v2 Patterns
from pydantic import BaseModel, TypeAdapter, ConfigDict
import time

class TelemetryEvent(BaseModel):
    # model_config controls validation behavior
    model_config = ConfigDict(
        frozen=True,          # Immutable after creation (hashable)
        strict=False,         # Allow coercion (str→float etc.)
        revalidate_instances="never",  # Skip re-validation on copy
        populate_by_name=True  # Accept both alias and field name
    )
    device_id:     str
    cpu_util:      float = Field(ge=0, le=100)
    anomaly_score: float = Field(ge=0, le=1)

# TypeAdapter: validate lists/dicts without a model wrapper
# Much faster than creating model instances for bulk validation
adapter = TypeAdapter(list[TelemetryEvent])

batch_data = [
    {"device_id": f"JNP-{i:03d}", "cpu_util": 45.0 + i, "anomaly_score": 0.1 * (i % 10)}
    for i in range(10_000)
]

t0 = time.perf_counter()
events = adapter.validate_python(batch_data)
elapsed = time.perf_counter() - t0

print(f"Validated {len(events):,} events in {elapsed*1000:.1f}ms")
print(f"Rate: {len(events)/elapsed:,.0f} events/sec")

# JSON serialization — also much faster with TypeAdapter
t0 = time.perf_counter()
json_bytes = adapter.dump_json(events)
elapsed = time.perf_counter() - t0
print(f"Serialized to JSON in {elapsed*1000:.1f}ms ({len(json_bytes)/1024:.0f} KB)")

CUSTOM VALIDATORS

Pydantic's built-in constraints handle simple cases. Custom validators handle business logic — cross-field rules, external lookups, domain invariants.

Python — field_validator
from pydantic import BaseModel, Field, field_validator
import re

class DeviceReport(BaseModel):
    device_id:  str
    risk_level: str
    score:      float = Field(ge=0, le=1)

    @field_validator("device_id")
    @classmethod
    def validate_device_id_format(cls, v: str) -> str:
        # Custom regex: must be like JNP-001 or MX-042
        if not re.match(r"^[A-Z]{2,4}-\d{3}$", v):
            raise ValueError(f"Invalid device ID: '{v}'. Expected format: JNP-001")
        return v.upper()   # Normalize to uppercase

    @field_validator("risk_level")
    @classmethod
    def normalize_risk(cls, v: str) -> str:
        # Accept case-insensitive and normalize
        normalized = v.upper().strip()
        if normalized not in {"LOW", "MED", "HIGH"}:
            raise ValueError(f"risk_level must be LOW/MED/HIGH, got '{v}'")
        return normalized

# field_validator normalizes input
r1 = DeviceReport(device_id="jnp-001", risk_level="high", score=0.9)
print(r1.device_id)   # "JNP-001" ← normalized
print(r1.risk_level)  # "HIGH" ← normalized

# Validator catches invalid format
try:
    DeviceReport(device_id="router-1", risk_level="HIGH", score=0.5)
except Exception as e:
    print(f"Caught: {e.errors()[0]['msg']}")

model_validator: Cross-Field Rules

Python — model_validator for Cross-Field Logic
from pydantic import BaseModel, Field, model_validator
from typing import Literal, Optional

class AlertReport(BaseModel):
    risk_level:         Literal["LOW", "MED", "HIGH"]
    score:              float = Field(ge=0, le=1)
    escalation_contact: Optional[str] = None
    actions:            list[str]
    notify_oncall:      bool = False

    @model_validator(mode="after")
    def high_risk_requires_escalation(self):
        # Cross-field rule: HIGH risk must have escalation
        if self.risk_level == "HIGH" and not self.escalation_contact:
            raise ValueError(
                "escalation_contact is required when risk_level is HIGH"
            )
        return self

    @model_validator(mode="after")
    def score_matches_risk(self):
        # Score should align with risk level
        thresholds = {"LOW": (0.0, 0.5), "MED": (0.4, 0.8), "HIGH": (0.7, 1.0)}
        lo, hi = thresholds[self.risk_level]
        if not (lo <= self.score <= hi):
            raise ValueError(
                f"score={self.score:.2f} inconsistent with risk_level={self.risk_level} "
                f"(expected {lo:.1f}–{hi:.1f})"
            )
        return self

    @model_validator(mode="after")
    def high_risk_sets_oncall(self):
        # Auto-set notify_oncall for HIGH risk
        if self.risk_level == "HIGH":
            self.notify_oncall = True
        return self

# Test cross-field validation
try:
    AlertReport(risk_level="HIGH", score=0.91, actions=["check"])
except Exception as e:
    print(f"Caught: {e.errors()[0]['msg']}")

# Valid — HIGH with escalation
alert = AlertReport(
    risk_level="HIGH", score=0.91,
    escalation_contact="oncall@marevlo.com",
    actions=["Restart rpd"]
)
print(f"notify_oncall auto-set: {alert.notify_oncall}")  # True

Computed Fields & Custom Serializers

Python — Computed Fields & Serializers
from pydantic import BaseModel, Field, computed_field, field_serializer
from typing import Literal
from datetime import datetime

class DeviceReport(BaseModel):
    device_id:   str
    risk_level:  Literal["LOW", "MED", "HIGH"]
    score:       float
    created_at:  datetime = Field(default_factory=datetime.utcnow)
    metrics:     dict[str, float] = Field(default_factory=dict)

    # Computed fields — derived from other fields, not stored
    @computed_field
    @property
    def severity_code(self) -> int:
        return {"LOW": 1, "MED": 2, "HIGH": 3}[self.risk_level]

    @computed_field
    @property
    def requires_immediate_action(self) -> bool:
        return self.risk_level == "HIGH" and self.score >= 0.85

    @computed_field
    @property
    def anomalous_metrics(self) -> list[str]:
        return [k for k, v in self.metrics.items() if v > 0.8]

    # Custom serializer — controls JSON output format
    @field_serializer("created_at")
    def serialize_dt(self, dt: datetime) -> str:
        return dt.strftime("%Y-%m-%dT%H:%M:%SZ")

    @field_serializer("score")
    def serialize_score(self, v: float) -> str:
        return f"{v:.3f}"  # Always 3 decimal places

report = DeviceReport(
    device_id="JNP-001", risk_level="HIGH", score=0.91,
    metrics={"cpu_util": 0.942, "mem_util": 0.673, "pfe_errors": 0.95}
)

print(f"Severity code: {report.severity_code}")           # 3
print(f"Immediate: {report.requires_immediate_action}")    # True
print(f"Anomalous: {report.anomalous_metrics}")            # ['cpu_util', 'pfe_errors']
print(report.model_dump_json(indent=2))

Async Validators & External Lookups

Python — Async Pydantic with External Validation
from pydantic import BaseModel, model_validator
from typing import ClassVar
import asyncio

# Pydantic doesn't natively support async validators,
# but here's the recommended pattern using __post_init__

class ValidatedDeviceReport(BaseModel):
    device_id: str
    risk_level: str

    # Registry of known device IDs (in prod: from DB/API)
    _known_devices: ClassVar[set[str]] = {"JNP-001", "JNP-002", "MX-001"}

    @model_validator(mode="after")
    def check_device_exists(self):
        if self.device_id not in self._known_devices:
            raise ValueError(f"Unknown device: {self.device_id}")
        return self

async def validate_with_external(data: dict) -> ValidatedDeviceReport:
    """Async pattern: validate externally, then construct Pydantic model."""
    # Step 1: Async external check (DB, API, cache)
    await asyncio.sleep(0)  # Simulate DB lookup
    known = {"JNP-001", "JNP-002", "MX-001"}

    if data["device_id"] not in known:
        raise ValueError(f"Device {data['device_id']} not in inventory")

    # Step 2: Construct validated model
    return ValidatedDeviceReport.model_validate(data)

async def main():
    good = await validate_with_external({"device_id": "JNP-001", "risk_level": "HIGH"})
    print(f"✓ Valid: {good.device_id}")

    try:
        await validate_with_external({"device_id": "UNKNOWN-999", "risk_level": "HIGH"})
    except ValueError as e:
        print(f"✗ Caught: {e}")

asyncio.run(main())

LLM OUTPUT VALIDATION

The complete pipeline: define a Pydantic model → convert to tool schema → call LLM → validate response → retry on failure.

The core pattern: use Pydantic to both define your tool schema AND validate the LLM's response. One model serves both purposes.

Python — Pydantic → Tool Schema → Validation
from pydantic import BaseModel, Field
from typing import Literal
import anthropic, json

class DeviceAnalysis(BaseModel):
    device_id:   str = Field(description="Analyzed device ID")
    risk_level:  Literal["LOW", "MED", "HIGH"] = Field(
        description="LOW=monitor, MED=schedule work, HIGH=immediate action"
    )
    score:       float = Field(ge=0, le=1, description="0=normal, 1=critical")
    root_cause:  str = Field(description="Single-sentence root cause")
    actions:     list[str] = Field(min_length=1, description="Ordered action items")

def model_to_tool(model: type[BaseModel], name: str, desc: str) -> dict:
    schema = model.model_json_schema()
    schema.pop("$defs", None); schema.pop("title", None)
    return {"name": name, "description": desc, "input_schema": schema}

client = anthropic.Anthropic()
tool = model_to_tool(DeviceAnalysis, "report_analysis", "Report device analysis findings")

response = client.messages.create(
    model="claude-opus-4-5", max_tokens=1024,
    tools=[tool],
    tool_choice={"type": "tool", "name": "report_analysis"},
    messages=[{"role": "user", "content":
        "JNP-001: CPU=94.2%, PFE errors=412/hr, BGP flaps=8. Analyze."}]
)

# Extract + validate in one step
raw = response.content[0].input
analysis = DeviceAnalysis.model_validate(raw)

print(f"Risk: {analysis.risk_level} | Score: {analysis.score:.2f}")
print(f"Cause: {analysis.root_cause}")
print(f"Actions: {len(analysis.actions)}")

Retry Loop with Error Feedback

Python — Validated LLM Call with Retry
from pydantic import BaseModel, ValidationError
import anthropic, json

def validated_llm_call(
    model: type[BaseModel],
    prompt: str,
    tool_name: str = "structured_output",
    max_retries: int = 3
) -> BaseModel:
    """Call Claude with automatic schema validation and retry."""
    client = anthropic.Anthropic()

    schema = model.model_json_schema()
    schema.pop("$defs", None); schema.pop("title", None)
    tool_def = {"name": tool_name, "description": "Return structured output", "input_schema": schema}

    current_prompt = prompt
    last_errors = []

    for attempt in range(max_retries):
        # Add error feedback on retry
        feedback = ""
        if last_errors:
            error_list = "\n".join(f"- {e}" for e in last_errors)
            feedback = f"\n\nYour previous response had validation errors:\n{error_list}\nPlease fix them."

        response = client.messages.create(
            model="claude-opus-4-5", max_tokens=1024,
            tools=[tool_def],
            tool_choice={"type": "tool", "name": tool_name},
            messages=[{"role": "user", "content": current_prompt + feedback}]
        )

        raw = response.content[0].input
        try:
            result = model.model_validate(raw)
            print(f"✓ Validated on attempt {attempt + 1}")
            return result
        except ValidationError as e:
            last_errors = [f"{err['loc']}: {err['msg']}" for err in e.errors()]
            print(f"✗ Attempt {attempt + 1} failed: {len(last_errors)} errors")

    raise RuntimeError(f"Failed after {max_retries} attempts. Last errors: {last_errors}")

# Usage
class DeviceReport(BaseModel):
    device_id:  str
    risk_level: Literal["LOW", "MED", "HIGH"]
    score:      float = Field(ge=0, le=1)
    actions:    list[str] = Field(min_length=1)

report = validated_llm_call(
    DeviceReport,
    "JNP-001: CPU=94.2%, PFE=412/hr. Provide structured analysis."

Async Batch Validation Pipeline

Python — Async Batch LLM Processing
import asyncio, anthropic
from pydantic import BaseModel, Field, ValidationError
from typing import Literal

class DeviceReport(BaseModel):
    device_id:  str
    risk_level: Literal["LOW", "MED", "HIGH"]
    score:      float = Field(ge=0, le=1)
    actions:    list[str]

async def analyze_device(
    device_id: str,
    telemetry: dict,
    semaphore: asyncio.Semaphore
) -> tuple[str, DeviceReport | Exception]:
    async with semaphore:  # Rate limiting
        try:
            client = anthropic.AsyncAnthropic()
            schema = DeviceReport.model_json_schema()
            schema.pop("$defs", None); schema.pop("title", None)

            response = await client.messages.create(
                model="claude-opus-4-5", max_tokens=512,
                tools=[{"name": "report", "description": "Report", "input_schema": schema}],
                tool_choice={"type": "tool", "name": "report"},
                messages=[{"role": "user", "content":
                    f"Analyze {device_id}: {telemetry}"}]
            )
            result = DeviceReport.model_validate(response.content[0].input)
            return device_id, result
        except Exception as e:
            return device_id, e

async def batch_analyze(devices: list[dict], max_concurrent: int = 5):
    semaphore = asyncio.Semaphore(max_concurrent)
    tasks = [analyze_device(d["id"], d, semaphore) for d in devices]
    results = await asyncio.gather(*tasks)

    successes = [(d, r) for d, r in results if not isinstance(r, Exception)]
    failures  = [(d, r) for d, r in results if isinstance(r, Exception)]

    print(f"✓ {len(successes)} successes | ✗ {len(failures)} failures")
    high_risk = [r for _, r in successes if r.risk_level == "HIGH"]
    print(f"HIGH risk devices: {[r.device_id for r in high_risk]}")
    return successes, failures

devices = [
    {"id": "JNP-001", "cpu": 94.2, "errors": 412},
    {"id": "JNP-002", "cpu": 45.0, "errors": 3},
    {"id": "JNP-003", "cpu": 78.3, "errors": 87},
]
asyncio.run(batch_analyze(devices))

Pydantic + Instructor at Scale

Python — Instrumented Instructor with Metrics
import instructor, anthropic, time
from pydantic import BaseModel, Field
from typing import Literal, TypeVar
from dataclasses import dataclass, field

T = TypeVar("T", bound=BaseModel)

@dataclass
class CallMetrics:
    total_calls:   int = 0
    total_retries: int = 0
    total_tokens:  int = 0
    latencies_ms:  list[float] = field(default_factory=list)

    @property
    def avg_latency_ms(self) -> float:
        return sum(self.latencies_ms) / len(self.latencies_ms) if self.latencies_ms else 0

    @property
    def retry_rate(self) -> float:
        return self.total_retries / self.total_calls if self.total_calls else 0

class InstrumentedExtractor:
    def __init__(self, max_retries: int = 3):
        self.client = instructor.from_anthropic(anthropic.Anthropic())
        self.max_retries = max_retries
        self.metrics = CallMetrics()

    def extract(self, model: type[T], prompt: str) -> T:
        t0 = time.perf_counter()
        self.metrics.total_calls += 1

        result, completion = self.client.messages.create_with_completion(
            model="claude-opus-4-5", max_tokens=1024,
            response_model=model,
            max_retries=self.max_retries,
            messages=[{"role": "user", "content": prompt}]
        )

        elapsed = (time.perf_counter() - t0) * 1000
        self.metrics.latencies_ms.append(elapsed)
        self.metrics.total_tokens += (
            completion.usage.input_tokens + completion.usage.output_tokens
        )
        return result

    def report(self):
        m = self.metrics
        print(f"Calls: {m.total_calls} | Tokens: {m.total_tokens:,} | "
              f"Avg latency: {m.avg_latency_ms:.0f}ms | Retry rate: {m.retry_rate:.0%}")

BAML INTRO

BAML (Boundary AI Markup Language) is a domain-specific language for defining LLM function signatures with full type safety, testing, and IDE support.

BAML treats LLM calls like typed function calls. You write a .baml file that defines: the output type (class/enum), the LLM to use, and the prompt template. BAML generates Python client code with full type safety.

BAML File Anatomy

Keyword
class
Defines a typed output structure. Like a Pydantic model but in BAML syntax.
Keyword
enum
Defines a set of allowed string values. Enforced at extraction time.
Keyword
function
An LLM call with typed inputs and typed output. The core BAML primitive.
Keyword
client
Configures which LLM provider and model to use for a function.
Keyword
prompt
The prompt template using Jinja2-style {{ variable }} syntax.
Keyword
test
Built-in test cases for your LLM functions — run with baml-cli test.
BAML — Complete Example: device_analysis.baml
// 1. Define output types

enum RiskLevel {
  LOW   @description("Normal operation, monitor only")
  MED   @description("Schedule maintenance within 48 hours")
  HIGH  @description("Immediate intervention required")
}

class DeviceReport {
  device_id    string     @description("Device identifier e.g. JNP-001")
  risk_level   RiskLevel
  score        float      @description("0.0=normal, 1.0=critical failure imminent")
  root_cause   string     @description("Single-sentence root cause")
  actions      string[]   @description("Ordered list of remediation steps")
}

// 2. Configure LLM client
client<llm> ClaudeOpus {
  provider   anthropic
  options {
    model       claude-opus-4-5
    max_tokens  1024
  }
}

// 3. Define the LLM function
function AnalyzeDevice(
  device_id: string,
  telemetry:  string
) -> DeviceReport {
  client ClaudeOpus
  prompt #"
    You are a network reliability engineer.

    Analyze device {{ device_id }} with this telemetry:
    {{ telemetry }}

    {{ ctx.output_format }}
  "#
}

// 4. Write tests (run with: baml-cli test)
test HighRiskDevice {
  functions [AnalyzeDevice]
  args {
    device_id  "JNP-001"
    telemetry  "CPU: 94.2%, PFE errors: 412/hr, BGP flaps: 8"
  }
}

class ValidatedReport {
  device_id    string   @check(this.matches("^[A-Z]{2,4}-\\d{3}$"), "Invalid device ID format")
  risk_level   RiskLevel
  score        float    @check(this >= 0 and this <= 1, "Score must be 0-1")
  actions      string[] @check(this.len() >= 1, "At least one action required")
  root_cause   string   @check(this.len() <= 200, "Root cause too long")

  // Cross-field check using a separate @assert block
  @assert(
    (risk_level == RiskLevel.HIGH and score >= 0.7) or
    (risk_level == RiskLevel.MED  and score >= 0.4 and score < 0.8) or
    (risk_level == RiskLevel.LOW  and score < 0.5),
    "Score must be consistent with risk_level"
  )
}

PRODUCTION PATTERNS

Hard-won lessons for Pydantic and BAML in production AI systems.

Pattern 1 — Model Hierarchy

Define three model levels: LLMOutput (minimal, permissive — what the LLM returns), DomainModel (full business validation), and APIResponse (serialization-optimized). Promote data up through the hierarchy as it's validated.

Pattern 2 — Never Validate on the Critical Path

Run Pydantic validation in a try/except and log failures to a dead-letter queue. Never let a validation error crash your main inference loop — degrade gracefully and alert async.

Pattern 3 — Schema-Driven Logging

Use model.model_dump_json() to serialize every LLM output to JSONL before validation. If validation later fails, you have the raw LLM output for debugging — without re-calling the API.

Python — Complete Production Pattern
from pydantic import BaseModel, Field, ValidationError
from typing import Literal, Optional
import instructor, anthropic, json, logging
from datetime import datetime, timezone

logger = logging.getLogger("hunter.pydantic")

# Level 1: Permissive LLM output model
class RawLLMOutput(BaseModel):
    device_id:  str
    risk_level: str            # Accept any string first
    score:      float | str    # Accept string or float
    actions:    list[str]

# Level 2: Domain-validated model
class DeviceReport(BaseModel):
    device_id:  str = Field(pattern=r"^[A-Z]{2,4}-\d{3}$")
    risk_level: Literal["LOW", "MED", "HIGH"]
    score:      float = Field(ge=0, le=1)
    actions:    list[str] = Field(min_length=1)
    validated_at: datetime = Field(default_factory=lambda: datetime.now(timezone.utc))

def safe_analyze(
    device_id: str,
    telemetry: dict,
    log_path: str = "/tmp/llm_outputs.jsonl"
) -> Optional[DeviceReport]:

    client = instructor.from_anthropic(anthropic.Anthropic())

    try:
        # Step 1: Get raw LLM output (permissive validation)
        raw: RawLLMOutput = client.messages.create(
            model="claude-opus-4-5", max_tokens=1024,
            response_model=RawLLMOutput, max_retries=3,
            messages=[{"role": "user", "content":
                f"Analyze {device_id}: {json.dumps(telemetry)}"}]
        )

        # Step 2: Log raw output BEFORE strict validation
        with open(log_path, "a") as f:
            f.write(raw.model_dump_json() + "\n")

        # Step 3: Strict domain validation
        validated = DeviceReport(
            device_id=raw.device_id.upper(),
            risk_level=raw.risk_level.upper(),
            score=float(raw.score),
            actions=raw.actions
        )
        logger.info(f"✓ {device_id}: {validated.risk_level} risk")
        return validated

    except ValidationError as e:
        logger.error(f"✗ Validation failed for {device_id}: {e.errors()}")
        return None  # Graceful degradation
    except Exception as e:
        logger.error(f"✗ API error for {device_id}: {e}")
        return None
