WHY TEMPLATES?

Python f-strings break on edge cases. Concatenation doesn't compose. Jinja2 gives you a complete, battle-tested templating engine — used in Flask, Ansible, Kubernetes, and now AI prompt engineering.

When you're building agents, your prompts are actually programs — they have logic, data injection, and conditional branches. Using Python f-strings for this is like writing SQL with string concatenation: it works until it doesn't, and then it fails in painful ways.

✗ F-STRING HELL
devices_str = ", ".join(devices)
examples_str = ""
for i, ex in enumerate(examples):
    examples_str += f"\nExample {i+1}:\n"
    examples_str += f"Input: {ex['input']}\n"
    examples_str += f"Output: {ex['output']}\n"

if include_history:
    history_str = f"\nHistory: {history}"
else:
    history_str = ""

prompt = f"""Analyze {devices_str}.
{examples_str}
{history_str}
Return JSON."""
# Breaks on: quotes in data, None values,
# multi-line strings, special chars...
✓ JINJA2 TEMPLATE
# template.j2
from jinja2 import Environment

TMPL = """
Analyze: {{ devices | join(", ") }}

{% for ex in examples %}
Example {{ loop.index }}:
  Input:  {{ ex.input }}
  Output: {{ ex.output | tojson }}
{% endfor %}

{% if include_history %}
History: {{ history }}
{% endif %}

Return JSON.
"""

env = Environment()
prompt = env.from_string(TMPL).render(
    devices=devices,
    examples=examples,
    include_history=True,
    history=history
)
>_

// TEMPLATE STRATEGY: SEPARATION OF CONCERNS

At the advanced level, Jinja2 templates enforce a clean separation between prompt structure (what the template knows) and runtime data (what Python injects). This separation enables prompt testing, versioning, and A/B testing independently of application logic.

Python — Template-as-Config Pattern

# For ultra high-throughput: pre-compile all templates at startup

    print(f"Pre-compiled {sum(1 for n in env.loader.list_templates() if n.endswith('.j2'))} templates")

JINJA2 SYNTAX BASICS

Three delimiter types control all Jinja2 behavior. Master these and you can write any template.

// THE THREE DELIMITERS

Expression
{{ variable }}
Outputs a value. Supports dot notation, filters, and method calls. {{ user.name }}, {{ items|length }}
Statement
{% block %}
Control structures: if/elif/else, for, set, block, macro, extends. Must have matching end tag.
Comment
{# comment #}
Completely removed from output. Use for template documentation and TODO notes.
Jinja2 — Core Syntax Reference
{# This is a comment — removed from output #}

{# 1. VARIABLE OUTPUT #}
{{ device_id }}                  {# Simple variable #}
{{ device.cpu_util }}            {# Dot notation (object or dict) #}
{{ devices[0] }}                 {# Index access #}
{{ device.get('cpu', 'N/A') }}   {# Method calls #}
{{ score | round(2) }}           {# Filter: round to 2 decimals #}
{{ name | upper | truncate(20) }} {# Chained filters #}

{# 2. CONDITIONALS #}
{% if risk_level == "HIGH" %}
🚨 IMMEDIATE ACTION REQUIRED
{% elif risk_level == "MED" %}
⚠️  Schedule maintenance
{% else %}
✓  Continue monitoring
{% endif %}

{# 3. FOR LOOPS #}
{% for device in devices %}
Device {{ loop.index }}: {{ device.id }}
  CPU: {{ device.cpu }}%
  {% if loop.last %}(last device){% endif %}
{% endfor %}

{# 4. SET VARIABLES #}
{% set threshold = 80 %}
{% set is_critical = score > 0.8 %}
{{ is_critical | string }}            {# "True" or "False" #}

Jinja2 — Loop Variables
{% for example in examples %}
{# loop.index:   1-based counter (1, 2, 3...) #}
{# loop.index0:  0-based counter (0, 1, 2...) #}

{# loop.last:    True on last iteration #}
{# loop.length:  Total number of items #}
{# loop.revindex: Reverse counter (n, n-1...) #}
{# loop.cycle('a','b'): Alternates values #}

=== Example {{ loop.index }} / {{ loop.length }} ===
Input:  {{ example.input }}
Output: {{ example.output | tojson(indent=2) }}
{% if not loop.last %}
---
{% endif %}
{% endfor %}

Jinja2 — Whitespace Trimming
{# Problem: block tags add blank lines #}
Device ID: {{ device_id }}
{% if include_history %}
History: {{ history }}
{% endif %}
{# ↑ Produces blank lines before/after the block #}

{# Solution 1: Dash trimming (-) #}
Device ID: {{ device_id }}
{%- if include_history %}
History: {{ history }}
{%- endif %}
{# ← Dash BEFORE tag removes preceding whitespace/newline #}
{# ← Dash AFTER tag would remove following whitespace/newline #}

{# Solution 2: Environment-level trimming (recommended) #}

# lstrip_blocks: removes leading whitespace before block tags #}

from jinja2 import Environment, FileSystemLoader, pass_context
from jinja2.ext import Extension
import hashlib, datetime

# Add custom global functions available in all templates
def make_env() -> Environment:
    env = Environment(
        loader=FileSystemLoader("prompts/"),
        trim_blocks=True,
        lstrip_blocks=True
    )

    # Global functions: accessible in templates as {{ fn() }}
    env.globals["now"] = lambda: datetime.datetime.utcnow().isoformat()
    env.globals["risk_emoji"] = lambda r: {"LOW": "✓", "MED": "⚠️", "HIGH": "🚨"}.get(r, "?")
    env.globals["enumerate"] = enumerate

    # Custom filters: {{ value | filter_name }}
    env.filters["truncate_smart"] = lambda s, n=100: s[:n] + "..." if len(s) > n else s
    env.filters["score_bar"] = lambda s: "█" * int(s * 10) + "░" * (10 - int(s * 10))
    env.filters["fingerprint"] = lambda s: hashlib.md5(str(s).encode()).hexdigest()[:8]

    return env

env = make_env()

Python + Jinja2 — AI-Specific Custom Filters
from jinja2 import Environment
import json, tiktoken

def make_ai_env() -> Environment:
    env = Environment(trim_blocks=True, lstrip_blocks=True)
    enc = tiktoken.encoding_for_model("gpt-4")

    # Count tokens of a string
    def token_count(text: str) -> int:
        return len(enc.encode(str(text)))

    # Truncate to a max token count
    def truncate_tokens(text: str, max_tokens: int = 500) -> str:
        tokens = enc.encode(str(text))
        if len(tokens) <= max_tokens:
            return text
        return enc.decode(tokens[:max_tokens]) + " [TRUNCATED]"

    # Format a score as a visual bar
    def score_bar(score: float, width: int = 10) -> str:
        filled = int(score * width)
        return "█" * filled + "░" * (width - filled)

    # Serialize to compact JSON (for embedding in prompts)
    def compact_json(obj) -> str:
        return json.dumps(obj, separators=(",", ":"), default=str)

    # Format telemetry dict as readable key-value pairs
    def fmt_telemetry(d: dict) -> str:
        return "\n".join(f"  {k}: {v}" for k, v in d.items())

    env.filters["token_count"] = token_count
    env.filters["truncate_tokens"] = truncate_tokens
    env.filters["score_bar"] = score_bar
    env.filters["compact_json"] = compact_json
    env.filters["fmt_telemetry"] = fmt_telemetry

    return env

# Usage in template:
# Score: {{ score | score_bar }}       → ████████░░
# Tokens: {{ long_doc | token_count }} → 342
# Doc: {{ doc | truncate_tokens(200) }} → first 200 tokens

Macros are callable template functions — like Python functions but for prompt snippets. Use them to DRY up repeated prompt patterns across your agent system.

A macro is defined once and called many times — perfect for few-shot example formatting, device report rendering, or any prompt snippet you reuse across multiple templates.

from jinja2 import Environment, FileSystemLoader

env = Environment(loader=FileSystemLoader("prompts/"), trim_blocks=True, lstrip_blocks=True)

def get_macro(env: Environment, template_name: str, macro_name: str):
    """Load and return a specific macro from a template."""
    module = env.get_template(template_name).module
    return getattr(module, macro_name)

# Load specific macros dynamically
device_summary = get_macro(env, "macros/components.j2", "device_summary")
few_shot = get_macro(env, "macros/components.j2", "few_shot_block")

# Call them like Python functions!
device = {"id": "JNP-001", "risk": "HIGH", "score": 0.91, "cpu": 94.2, "mem": 67.3, "errors": 412}
summary_html = device_summary(device, show_actions=False)

# Dynamic macro selection based on task type
TASK_MACROS = {
    "analysis":      ("macros/analysis.j2", "device_summary"),
    "classification": ("macros/classify.j2", "risk_classifier"),
    "rag_qa":         ("macros/rag.j2",      "qa_block"),
}

def render_task(task_type: str, **kwargs) -> str:
    tmpl_name, macro_name = TASK_MACROS[task_type]
    macro = get_macro(env, tmpl_name, macro_name)
    return macro(**kwargs)

TEMPLATE INHERITANCE

Template inheritance lets child templates override specific blocks of a base template — like object inheritance but for prompts. Critical for multi-agent systems with shared prompt structure.

Inheritance works like this: a base template defines the overall structure with named {% block %} placeholders. Child templates extend the base and override specific blocks while inheriting everything else.

base_agent.j2
{# Base template: shared structure #}

{% block role %}
<role>You are an AI agent.</role>
{% endblock %}

{% block context %}{% endblock %}

{% block data %}{% endblock %}

{% block task %}
Complete the task above.
{% endblock %}

{% block format %}
Return JSON.
{% endblock %}
device_analyst.j2
{# Child: overrides specific blocks #}
{% extends "base_agent.j2" %}

{% block role %}
<role>Senior network reliability
engineer specializing in Juniper
router diagnostics.</role>
{% endblock %}

{% block data %}
<telemetry device="{{ device_id }}">
  CPU:    {{ cpu_util }}%
  Errors: {{ pfe_errors }}/hr
</telemetry>
{% endblock %}

{% block task %}
Determine risk level and actions.
{% endblock %}
{# format block inherited as-is #}
>_
super() call: In a child block, use {{ super() }} to render the parent block's content and add to it — rather than completely replacing it. Essential for adding constraints without rewriting the entire role definition.

{{ super() }}  {# Include parent role #}

Python — Dynamic Template Assembly
from jinja2 import Environment, FileSystemLoader, DictLoader
from typing import Optional

class PromptTemplateComposer:
    """Assembles prompt templates dynamically at runtime."""

    def __init__(self, template_dir: str):
        self.file_env = Environment(
            loader=FileSystemLoader(template_dir),
            trim_blocks=True, lstrip_blocks=True
        )

    def compose(
        self,
        base: str,                      # Base template name
        overrides: dict[str, str],      # block_name → content
        **render_kwargs
    ) -> str:
        """Dynamically compose a template by overriding specific blocks."""
        # Build a dynamic child template that extends base
        block_strs = "\n".join(
            f"""{{% block {name} %}}\n{content}\n{{% endblock %}}"""
            for name, content in overrides.items()
        )
        child_source = f"""{{% extends "{base}" %}}\n{block_strs}"""

        # Create a temporary environment with the dynamic template
        templates = {"__dynamic__": child_source}
        # Merge with file loader
        mixed_env = Environment(
            loader=DictLoader(templates),
            trim_blocks=True, lstrip_blocks=True
        )
        # Copy templates from file loader
        mixed_env.loader.mapping.update({
            name: self.file_env.loader.get_source(self.file_env, name)[0]
            for name in self.file_env.loader.list_templates()
        })

        return mixed_env.get_template("__dynamic__").render(**render_kwargs)

# Usage
composer = PromptTemplateComposer("prompts/")
prompt = composer.compose(
    base="base_agent.j2",
    overrides={
        "role": "You are a specialized routing protocol expert.",
        "task": "Analyze BGP session stability for device {{ device_id }}.",
    },
    device_id="JNP-001"
)

from jinja2 import Environment
from dataclasses import dataclass
import anthropic, json, random

@dataclass
class FewShotExample:
    device_id:  str
    telemetry:  dict
    risk_level: str
    score:      float
    actions:    list[str]

FEW_SHOT_TEMPLATE = """
You are a network reliability engineer.

=== EXAMPLES ===
{% for ex in examples %}
[Example {{ loop.index }}]
Device: {{ ex.device_id }}
Telemetry:
{% for key, val in ex.telemetry.items() %}
  {{ key }}: {{ val }}
{% endfor %}
Expected Output: {{ {"risk_level": ex.risk_level, "score": ex.score, "actions": ex.actions} | tojson }}
{% if not loop.last %}---{% endif %}
{% endfor %}

=== YOUR TASK ===
Device: {{ device_id }}
Telemetry:
{% for key, val in telemetry.items() %}
  {{ key }}: {{ val }}
{% endfor %}

Analyze this device and return JSON matching the examples above.
"""

def build_few_shot_prompt(
    device_id: str,
    telemetry: dict,
    example_bank: list[FewShotExample],
    n_examples: int = 3,
    strategy: str = "diverse"
) -> str:
    if strategy == "diverse":
        # Pick one from each risk level
        by_risk = {}
        for ex in example_bank:
            by_risk.setdefault(ex.risk_level, []).append(ex)
        selected = []
        for risk in ["LOW", "MED", "HIGH"]:
            if risk in by_risk:
                selected.append(random.choice(by_risk[risk]))
    else:  # random
        selected = random.sample(example_bank, min(n_examples, len(example_bank)))

    env = Environment(trim_blocks=True, lstrip_blocks=True)
    return env.from_string(FEW_SHOT_TEMPLATE).render(
        device_id=device_id, telemetry=telemetry, examples=selected
    )

Python — Automated Template Eval
from jinja2 import Environment, FileSystemLoader
from pydantic import BaseModel, Field
from typing import Literal
import instructor, anthropic, json
from dataclasses import dataclass, field

@dataclass
class EvalCase:
    name:            str
    template_vars:   dict
    expected_risk:   str
    min_score:       float = 0.0

class EvalOutput(BaseModel):
    risk_level: Literal["LOW", "MED", "HIGH"]
    score:      float = Field(ge=0, le=1)

def eval_template(
    template_path: str,
    cases: list[EvalCase],
    verbose: bool = True
) -> dict:
    env = Environment(loader=FileSystemLoader("prompts/"), trim_blocks=True, lstrip_blocks=True)
    tmpl = env.get_template(template_path)
    client = instructor.from_anthropic(anthropic.Anthropic())

    results = {"passed": 0, "failed": 0, "details": []}

    for case in cases:
        prompt = tmpl.render(**case.template_vars)
        output: EvalOutput = client.messages.create(
            model="claude-opus-4-5", max_tokens=256,
            response_model=EvalOutput,
            messages=[{"role": "user", "content": prompt}]
        )
        risk_ok  = output.risk_level == case.expected_risk
        score_ok = output.score >= case.min_score
        passed   = risk_ok and score_ok

        results["passed" if passed else "failed"] += 1
        results["details"].append({
            "case": case.name, "passed": passed,
            "got_risk": output.risk_level, "expected_risk": case.expected_risk,
            "score": output.score
        })

        if verbose:
            status = "✓ PASS" if passed else "✗ FAIL"
            print(f"  {status} {case.name}: risk={output.risk_level} score={output.score:.2f}")

    results["accuracy"] = results["passed"] / len(cases)
    return results

// PATTERN 1 — SANDBOX UNTRUSTED INPUT

Never render user-provided template strings. Use SandboxedEnvironment if you must allow dynamic templates. Always validate template vars before rendering — especially if they come from user input or LLM outputs.

from jinja2.sandbox import SandboxedEnvironment

# SandboxedEnvironment restricts dangerous operations

safe_env = SandboxedEnvironment(trim_blocks=True)

def safe_render_user_template(template_str: str, variables: dict) -> str:
    """Safely render a user-provided template string."""

// PATTERN 2 — TOKEN BUDGET ENFORCEMENT
Python — Token-Aware Template Rendering
import tiktoken
from jinja2 import Environment

enc = tiktoken.encoding_for_model("gpt-4")

def render_with_budget(
    env: Environment,
    template_name: str,
    variables: dict,
    max_tokens: int = 8000,
    truncation_strategy: str = "trim_documents"
) -> tuple[str, int]:
    """Render template and enforce token budget."""
    tmpl = env.get_template(template_name)
    prompt = tmpl.render(**variables)
    tokens = len(enc.encode(prompt))

    if tokens <= max_tokens:
        return prompt, tokens

    # Over budget — apply truncation strategy
    if truncation_strategy == "trim_documents" and "documents" in variables:
        # Progressively shorten documents until under budget
        docs = variables["documents"]
        while tokens > max_tokens and docs:
            docs = [{**d, "content": d["content"][:len(d["content"]) // 2]} for d in docs]
            prompt = tmpl.render(**{**variables, "documents": docs})
            tokens = len(enc.encode(prompt))

    elif truncation_strategy == "trim_examples" and "examples" in variables:
        # Remove examples one by one until under budget
        examples = variables["examples"][:]
        while tokens > max_tokens and len(examples) > 1:
            examples.pop()
            prompt = tmpl.render(**{**variables, "examples": examples})
            tokens = len(enc.encode(prompt))

    return prompt, tokens

Python — Production Template Registry
from jinja2 import Environment, FileSystemLoader
from threading import Lock
from pathlib import Path
import hashlib, time

class TemplateRegistry:
    """Thread-safe template registry with hot-reload support."""

    def __init__(self, template_dir: str, auto_reload: bool = False):
        self.env = Environment(
            loader=FileSystemLoader(template_dir),
            trim_blocks=True, lstrip_blocks=True,
            auto_reload=auto_reload   # True in dev, False in prod
        )
        self._checksums: dict[str, str] = {}
        self._lock = Lock()
        self._render_count = 0
        self._error_count = 0

    def render(self, template_name: str, **kwargs) -> str:
        with self._lock:
            self._render_count += 1
        try:
            return self.env.get_template(template_name).render(**kwargs)
        except Exception as e:
            with self._lock:
                self._error_count += 1
            raise

    def validate_all(self) -> list[str]:
        """Validate all templates at startup — catch syntax errors early."""
        errors = []
        for name in self.env.loader.list_templates():
            try:
                self.env.get_template(name)
            except Exception as e:
                errors.append(f"{name}: {e}")
        return errors

    def health(self) -> dict:
        return {
            "renders": self._render_count,
            "errors": self._error_count,
            "error_rate": self._error_count / max(1, self._render_count)
        }

# Singleton pattern — one registry per process
registry = TemplateRegistry("prompts/", auto_reload=False)

# Validate at startup
errors = registry.validate_all()
if errors:
    raise RuntimeError(f"Template errors found: {errors}")

# Use throughout app
prompt = registry.render("device_analysis.j2", device_id="JNP-001", cpu=94.2)
